```rust
impl JitCompiler {
    fn compile_value_operation(&self, op: ValueOp) -> CompiledCode {
        match op {
            ValueOp::Add(a, b) if self.can_specialize_numeric(a, b) => {
                self.generate_specialized_add(a, b)
            }
            _ => self.generate_generic_operation(op)
        }
    }
}
```
