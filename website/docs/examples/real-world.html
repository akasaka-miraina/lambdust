Web Server with Concurrent Request Handling

A high-performance HTTP server showcasing Lambdust's concurrent programming capabilities:

http-server.ldust

;; Production HTTP server with concurrent request handling
(import (scheme base)
        (lambdust web http)
        (lambdust concurrency)
        (lambdust json)
        (lambdust time))

;; Request handler types
(define-type Request
  (Request (method : HttpMethod)
          (path : String)
          (headers : (Map String String))
          (body : (Maybe String))))

(define-type Response
  (Response (status : Integer)
           (headers : (Map String String))
           (body : String)))

;; Application state
(define-type AppState
  (AppState (request-count : (Atomic Integer))
           (active-connections : (Atomic Integer))
           (start-time : Timestamp)))

;; Initialize application state
(define app-state : AppState
  (AppState (make-atomic 0)
           (make-atomic 0)
           (current-timestamp)))

;; Middleware for logging requests
(define request-logger : (Request → IO Request)
  (lambda ((req : Request))
    (do-io
      [count ← (atomic-increment! (request-count app-state))]
      [timestamp ← (current-timestamp)]
      [_ ← (log-info (format "~a ~a [~a] Request #~a"
                            (method req)
                            (path req)
                            (format-timestamp timestamp)
                            count))]
      (return req))))

;; JSON API endpoints
(define api-routes : (Map String (Request → IO Response))
  (hash-map
    '(("/api/status" . handle-status)
      ("/api/health" . handle-health)
      ("/api/users" . handle-users)
      ("/api/stats" . handle-stats))))

;; Status endpoint
(define handle-status : (Request → IO Response)
  (lambda ((req : Request))
    (let ([status-data (json-object
                         '("status" "ok")
                         '("timestamp" (timestamp->string (current-timestamp)))
                         '("version" "1.0.0"))])
      (return (Response 200
                       (hash-map '(("Content-Type" . "application/json")))
                       (json->string status-data))))))

;; Health check endpoint
(define handle-health : (Request → IO Response)
  (lambda ((req : Request))
    (do-io
      [req-count ← (atomic-read (request-count app-state))]
      [connections ← (atomic-read (active-connections app-state))]
      [uptime ← (timestamp-diff (current-timestamp) (start-time app-state))]
      [memory-usage ← (get-memory-usage)]
      [health-data (json-object
                     '("healthy" #t)
                     '("uptime_seconds" uptime)
                     '("requests_served" req-count)
                     '("active_connections" connections)
                     '("memory_mb" (/ memory-usage 1048576)))]
      (return (Response 200
                       (hash-map '(("Content-Type" . "application/json")))
                       (json->string health-data))))))

;; User management (with database integration)
(define handle-users : (Request → IO Response)
  (lambda ((req : Request))
    (match (method req)
      ['GET (get-all-users)]
      ['POST (create-user req)]
      ['PUT (update-user req)]
      ['DELETE (delete-user req)]
      [_ (return (Response 405
                          (hash-map '(("Allow" . "GET, POST, PUT, DELETE")))
                          "Method not allowed"))])))

;; Database connection pool
(define-type DbPool
  (DbPool (connections : (Channel DbConnection))
         (max-size : Integer)
         (current-size : (Atomic Integer))))

(define db-pool : DbPool
  (DbPool (make-channel 10)
         10
         (make-atomic 0)))

;; Safe database operation with connection pooling
(define with-db-connection : (∀ α. (DbConnection → IO α) → IO α)
  (lambda (operation)
    (do-io
      [conn ← (channel-receive (connections db-pool))]
      [result ← (catch-errors (operation conn)
                              (lambda (error)
                                (log-error (format "Database error: ~a" error))
                                (return (Left error))))]
      [_ ← (channel-send (connections db-pool) conn)]
      (match result
        [(Right value) (return value)]
        [(Left error) (raise error)]))))

;; Get all users with pagination
(define get-all-users : (IO Response)
  (lambda ()
    (with-db-connection
      (lambda (conn)
        (do-io
          [users ← (db-query conn "SELECT id, name, email FROM users LIMIT 100")]
          [user-list (map (lambda (row)
                           (json-object
                             '("id" (row-ref row "id"))
                             '("name" (row-ref row "name"))
                             '("email" (row-ref row "email"))))
                         users)]
          [response-data (json-object '("users" user-list)
                                     '("count" (length users)))]
          (return (Response 200
                           (hash-map '(("Content-Type" . "application/json")))
                           (json->string response-data))))))))

;; Request routing with pattern matching
(define route-request : (Request → IO Response)
  (lambda ((req : Request))
    (match (path req)
      [path (when (hash-map-contains? api-routes path)) =>
       (lambda ()
         (let ([handler (hash-map-ref api-routes path)])
           (handler req)))]
      ["/favicon.ico" =>
       (lambda ()
         (return (Response 404 (hash-map) "Not found")))]
      ["/" =>
       (lambda ()
         (serve-static-file "static/index.html"))]
      [path (when (string-prefix? "/static/" path)) =>
       (lambda ()
         (serve-static-file (string-drop path 1)))]
      [_ =>
       (lambda ()
         (return (Response 404
                          (hash-map '(("Content-Type" . "application/json")))
                          "{\"error\": \"Not found\"}")))])))

;; Concurrent request handler
(define handle-request : (Socket → IO Unit)
  (lambda ((client-socket : Socket))
    (do-io
      [_ ← (atomic-increment! (active-connections app-state))]
      [result ← (catch-errors
                  (do-io
                    [request ← (parse-http-request client-socket)]
                    [logged-request ← (request-logger request)]
                    [response ← (route-request logged-request)]
                    [_ ← (send-http-response client-socket response)]
                    (return "success"))
                  (lambda (error)
                    (do-io
                      [_ ← (log-error (format "Request handling error: ~a" error))]
                      [error-response (Response 500
                                               (hash-map '(("Content-Type" . "text/plain")))
                                               "Internal server error")]
                      [_ ← (send-http-response client-socket error-response)]
                      (return "error"))))]
      [_ ← (socket-close client-socket)]
      [_ ← (atomic-decrement! (active-connections app-state))]
      (return unit))))

;; Main server loop with thread pool
(define start-server : (Integer → IO Unit)
  (lambda ((port : Integer))
    (do-io
      [server-socket ← (socket-listen port)]
      [thread-pool ← (make-thread-pool 20)]  ; 20 worker threads
      [_ ← (log-info (format "Server starting on port ~a" port))]
      (let server-loop ()
        (do-io
          [client-socket ← (socket-accept server-socket)]
          [_ ← (thread-pool-submit thread-pool
                                   (lambda () (handle-request client-socket)))]
          (server-loop))))))

;; Configuration and startup
(define main : (IO Unit)
  (lambda ()
    (do-io
      [config ← (load-config "config.json")]
      [port (json-ref config "port" 8080)]
      [_ ← (init-logging (json-ref config "log_level" "info"))]
      [_ ← (init-database db-pool (json-ref config "database_url"))]
      [_ ← (log-info "Starting Lambdust HTTP server...")]
      (start-server port))))

;; Error handling and graceful shutdown
(with-signal-handler 'SIGINT
  (lambda (signal)
    (do-io
      [_ ← (log-info "Shutting down server...")]
      [_ ← (cleanup-database db-pool)]
      [_ ← (log-info "Server shutdown complete")]
      (exit 0)))
  main)

Data Processing Pipeline

A data processing system using Lambdust's monadic effects and parallel computing:

data-pipeline.ldust

;; High-performance data processing pipeline
(import (scheme base)
        (lambdust streaming)
        (lambdust parallel)
        (lambdust csv)
        (lambdust json)
        (lambdust statistics))

;; Data types for our pipeline
(define-type DataRecord
  (DataRecord (id : String)
             (timestamp : Timestamp)
             (value : Real)
             (category : String)
             (metadata : (Map String Dynamic))))

(define-type ProcessingResult
  (ProcessingResult (total-records : Integer)
                   (processed : Integer)
                   (errors : Integer)
                   (statistics : DataStatistics)))

(define-type DataStatistics
  (DataStatistics (mean : Real)
                 (median : Real)
                 (std-dev : Real)
                 (min : Real)
                 (max : Real)
                 (categories : (Map String Integer))))

;; Streaming data parser with error recovery
(define parse-csv-record : (String → Either String DataRecord)
  (lambda ((line : String))
    (catch-parse-errors
      (let ([fields (csv-parse-line line)])
        (if (= (length fields) 5)
            (Right (DataRecord (list-ref fields 0)
                              (parse-timestamp (list-ref fields 1))
                              (string->number (list-ref fields 2))
                              (list-ref fields 3)
                              (json-parse (list-ref fields 4))))
            (Left (format "Invalid field count: expected 5, got ~a" (length fields)))))
      (lambda (error)
        (Left (format "Parse error: ~a" error))))))

;; Data validation with dependent types
(define validate-record : (DataRecord → Maybe DataRecord)
  (lambda ((record : DataRecord))
    (cond
      [(not (> (timestamp record) (timestamp "2020-01-01")))
       None]  ; Too old
      [(not (and (>= (value record) 0) (<= (value record) 1000000)))
       None]  ; Value out of range
      [(string-empty? (category record))
       None]  ; Missing category
      [else (Some record)])))

;; Parallel data transformation
(define transform-record : (DataRecord → DataRecord)
  (lambda ((record : DataRecord))
    (DataRecord (id record)
               (timestamp record)
               ;; Normalize value to 0-1 range
               (/ (value record) 1000000.0)
               (string-downcase (category record))
               ;; Add computed metadata
               (hash-map-set (metadata record)
                            "processed_at"
                            (timestamp->string (current-timestamp))))))

;; Streaming aggregation with state
(define-type AggregationState
  (AggregationState (count : Integer)
                   (sum : Real)
                   (sum-squares : Real)
                   (min-val : Real)
                   (max-val : Real)
                   (values : (List Real))
                   (categories : (Map String Integer))))

(define initial-state : AggregationState
  (AggregationState 0 0.0 0.0 #:infinity #:-infinity '() (hash-map)))

(define update-aggregation : (AggregationState → DataRecord → AggregationState)
  (lambda ((state : AggregationState) (record : DataRecord))
    (let ([val (value record)]
          [cat (category record)])
      (AggregationState
        (+ (count state) 1)
        (+ (sum state) val)
        (+ (sum-squares state) (* val val))
        (min (min-val state) val)
        (max (max-val state) val)
        (cons val (values state))
        (hash-map-update (categories state) cat
                        (lambda (count) (+ count 1))
                        0)))))

;; Compute final statistics
(define compute-statistics : (AggregationState → DataStatistics)
  (lambda ((state : AggregationState))
    (let ([n (count state)]
          [mean (/ (sum state) n)]
          [variance (- (/ (sum-squares state) n) (* mean mean))])
      (DataStatistics
        mean
        (list-median (values state))
        (sqrt variance)
        (min-val state)
        (max-val state)
        (categories state)))))

;; Main processing pipeline
(define process-data-file : (String → String → IO ProcessingResult)
  (lambda ((input-file : String) (output-file : String))
    (do-io
      [_ ← (log-info (format "Processing file: ~a" input-file))]
      [start-time ← (current-timestamp)]
      
      ;; Create processing pipeline
      [result ← (with-file-input input-file
                  (lambda (input-stream)
                    (with-file-output output-file
                      (lambda (output-stream)
                        ;; Streaming pipeline with parallel processing
                        (streaming-fold
                          ;; Input: lines from CSV file
                          (stream-lines input-stream)
                          
                          ;; Initial accumulator
                          (tuple initial-state 0 0)  ; (state, processed, errors)
                          
                          ;; Processing function
                          (lambda (acc line)
                            (match acc
                              [(tuple state processed errors) =>
                               (lambda ()
                                 (match (parse-csv-record line)
                                   [(Right record) =>
                                    (lambda ()
                                      (match (validate-record record)
                                        [(Some valid-record) =>
                                         (lambda ()
                                           (let ([transformed (transform-record valid-record)])
                                             ;; Write to output file
                                             (write-json-line output-stream transformed)
                                             ;; Update aggregation
                                             (tuple (update-aggregation state transformed)
                                                   (+ processed 1)
                                                   errors)))]
                                        [None =>
                                         (lambda ()
                                           (log-warning (format "Invalid record: ~a" (id record)))
                                           (tuple state processed (+ errors 1)))]))]
                                   [(Left error) =>
                                    (lambda ()
                                      (log-error (format "Parse error: ~a" error))
                                      (tuple state processed (+ errors 1)))]))]))))))]
      
      [end-time ← (current-timestamp)]
      [duration (timestamp-diff end-time start-time)]
      
      (match result
        [(tuple final-state processed errors) =>
         (lambda ()
           (let ([stats (compute-statistics final-state)])
             (do-io
               [_ ← (log-info (format "Processing complete: ~a records in ~a seconds"
                                     processed duration))]
               [_ ← (log-info (format "Errors: ~a" errors))]
               [_ ← (log-info (format "Mean: ~a, Std Dev: ~a" 
                                     (mean stats) (std-dev stats)))]
               (return (ProcessingResult (+ processed errors)
                                        processed
                                        errors
                                        stats)))))]))))

;; Parallel batch processing
(define process-multiple-files : (List String → String → IO (List ProcessingResult))
  (lambda ((input-files : List String) (output-dir : String))
    (parallel-map
      (lambda (input-file)
        (let ([output-file (path-join output-dir
                                     (string-append
                                       (path-basename input-file)
                                       ".processed.json"))])
          (process-data-file input-file output-file)))
      input-files)))

;; Real-time streaming processor
(define start-stream-processor : (String → String → IO Unit)
  (lambda ((input-stream-url : String) (output-stream-url : String))
    (do-io
      [input-stream ← (connect-stream input-stream-url)]
      [output-stream ← (connect-stream output-stream-url)]
      [processor-state ← (make-atomic initial-state)]
      
      ;; Process incoming records continuously
      (stream-forever
        (do-io
          [raw-data ← (stream-read input-stream)]
          [result ← (async
                      (match (parse-csv-record raw-data)
                        [(Right record) =>
                         (lambda ()
                           (match (validate-record record)
                             [(Some valid-record) =>
                              (lambda ()
                                (let ([transformed (transform-record valid-record)])
                                  (do-io
                                    [_ ← (stream-write output-stream
                                                      (json->string transformed))]
                                    [_ ← (atomic-update! processor-state
                                                        (lambda (state)
                                                          (update-aggregation state transformed)))]
                                    (return "processed"))))]
                             [None =>
                              (lambda ()
                                (log-warning "Invalid record")
                                (return "skipped"))]))]
                        [(Left error) =>
                         (lambda ()
                           (log-error (format "Parse error: ~a" error))
                           (return "error"))]))]
          [_ ← (await result)]
          (return unit))))))

;; Command-line interface
(define main : (List String → IO Unit)
  (lambda ((args : List String))
    (match args
      [("batch" input-file output-file) =>
       (lambda ()
         (do-io
           [result ← (process-data-file input-file output-file)]
           [_ ← (display-result result)]
           (return unit)))]
      
      [("multi" input-dir output-dir) =>
       (lambda ()
         (do-io
           [input-files ← (directory-list input-dir "*.csv")]
           [results ← (process-multiple-files input-files output-dir)]
           [_ ← (map display-result results)]
           (return unit)))]
      
      [("stream" input-url output-url) =>
       (lambda ()
         (start-stream-processor input-url output-url))]
      
      [_ =>
       (lambda ()
         (do-io
           [_ ← (display "Usage:")]
           [_ ← (display "  lambdust data-pipeline.ldust batch  ")]
           [_ ← (display "  lambdust data-pipeline.ldust multi  ")]
           [_ ← (display "  lambdust data-pipeline.ldust stream  ")]
           (return unit)))])))

;; Helper to display results
(define display-result : (ProcessingResult → IO Unit)
  (lambda ((result : ProcessingResult))
    (do-io
      [_ ← (display (format "Total records: ~a" (total-records result)))]
      [_ ← (display (format "Successfully processed: ~a" (processed result)))]
      [_ ← (display (format "Errors: ~a" (errors result)))]
      [stats (statistics result)]
      [_ ← (display (format "Statistics: mean=~a, std=~a"
                           (mean stats) (std-dev stats)))]
      (return unit))))

Distributed Computing System

A distributed computation framework leveraging Lambdust's actor model and fault tolerance:

distributed-compute.ldust

;; Distributed computing system with fault tolerance
(import (scheme base)
        (lambdust actors)
        (lambdust network)
        (lambdust serialization)
        (lambdust time))

;; Message types for distributed system
(define-type Message
  (WorkRequest (id : String) (task : Task) (requester : ActorRef))
  (WorkResponse (id : String) (result : TaskResult))
  (HeartBeat (node-id : String) (timestamp : Timestamp))
  (NodeJoin (node-id : String) (address : NetworkAddress))
  (NodeLeave (node-id : String))
  (Redistribute (failed-node : String) (pending-tasks : (List WorkRequest))))

;; Task representation
(define-type Task
  (MapTask (data : (List α)) (mapper : (α → β)))
  (ReduceTask (data : (List α)) (reducer : (α → α → α)) (initial : α))
  (ComputeTask (computation : (Unit → α))))

;; Cluster node actor
(define node-actor : (String → NetworkAddress → Behavior)
  (lambda ((node-id : String) (address : NetworkAddress))
    (letrec ([healthy-nodes (make-hash-map)]
             [pending-tasks (make-hash-map)]
             [completed-tasks (make-hash-map)]
             
             [handle-message
               (lambda (message sender state)
                 (match message
                   [(WorkRequest id task requester) =>
                    (lambda ()
                      ;; Execute task asynchronously
                      (spawn-task
                        (lambda ()
                          (let ([result (execute-task task)])
                            (send! requester (WorkResponse id result))))
                      ;; Track pending task
                      (hash-map-set! pending-tasks id (tuple task requester))
                      state)]
                   
                   [(WorkResponse id result) =>
                    (lambda ()
                      ;; Mark task as completed
                      (hash-map-remove! pending-tasks id)
                      (hash-map-set! completed-tasks id result)
                      (log-info (format "Task ~a completed" id))
                      state)]
                   
                   [(HeartBeat from-node timestamp) =>
                    (lambda ()
                      ;; Update node health status
                      (hash-map-set! healthy-nodes from-node timestamp)
                      ;; Clean up old heartbeats
                      (cleanup-stale-nodes!)
                      state)]
                   
                   [(NodeJoin new-node new-address) =>
                    (lambda ()
                      (log-info (format "Node ~a joined cluster" new-node))
                      (hash-map-set! healthy-nodes new-node (current-timestamp))
                      ;; Redistribute load if needed
                      (maybe-redistribute-tasks!)
                      state)]
                   
                   [(NodeLeave failed-node) =>
                    (lambda ()
                      (log-warning (format "Node ~a left cluster" failed-node))
                      (hash-map-remove! healthy-nodes failed-node)
                      ;; Redistribute failed node's tasks
                      (redistribute-failed-tasks! failed-node)
                      state)]
                   
                   [_ =>
                    (lambda ()
                      (log-warning (format "Unknown message: ~a" message))
                      state)]))]
             
             [cleanup-stale-nodes!
               (lambda ()
                 (let ([current-time (current-timestamp)]
                       [timeout-threshold 30]) ; 30 seconds
                   (hash-map-for-each
                     healthy-nodes
                     (lambda (node-id last-seen)
                       (when (> (timestamp-diff current-time last-seen) timeout-threshold)
                         (log-warning (format "Node ~a appears dead" node-id))
                         (hash-map-remove! healthy-nodes node-id)
                         (redistribute-failed-tasks! node-id))))))]
             
             [redistribute-failed-tasks!
               (lambda (failed-node)
                 ;; Find tasks that were assigned to failed node
                 (let ([failed-tasks (filter
                                       (lambda (task-entry)
                                         (match task-entry
                                           [(id . (tuple task requester)) =>
                                            (lambda ()
                                              ;; Check if this task was on failed node
                                              (string=? (actor-node requester) failed-node))]))
                                       (hash-map->list pending-tasks))])
                   ;; Reassign to healthy nodes
                   (for-each
                     (lambda (task-entry)
                       (match task-entry
                         [(id . (tuple task original-requester)) =>
                          (lambda ()
                            (let ([target-node (select-least-loaded-node)])
                              (when target-node
                                (send! target-node (WorkRequest id task self))
                                (log-info (format "Redistributed task ~a to ~a"
                                                 id (actor-id target-node))))))]))
                     failed-tasks)))]
             
             [select-least-loaded-node
               (lambda ()
                 ;; Simple load balancing: pick node with fewest pending tasks
                 (fold-left
                   (lambda (best-node current-node-entry)
                     (match current-node-entry
                       [(node-id . last-seen) =>
                        (lambda ()
                          (let ([node-load (count-node-tasks node-id)])
                            (if (or (not best-node)
                                   (< node-load (count-node-tasks (car best-node))))
                                current-node-entry
                                best-node)))]))
                   #f
                   (hash-map->list healthy-nodes)))]
             
             [count-node-tasks
               (lambda (node-id)
                 (length (filter
                           (lambda (task-entry)
                             (match task-entry
                               [(id . (tuple task requester)) =>
                                (lambda ()
                                  (string=? (actor-node requester) node-id))]))
                           (hash-map->list pending-tasks))))])
      
      ;; Return behavior function
      (behavior handle-message (hash-map)))))

;; Task execution with error handling
(define execute-task : (Task → TaskResult)
  (lambda ((task : Task))
    (catch-errors
      (match task
        [(MapTask data mapper) =>
         (lambda ()
           (Success (map mapper data)))]
        
        [(ReduceTask data reducer initial) =>
         (lambda ()
           (Success (fold-left reducer initial data)))]
        
        [(ComputeTask computation) =>
         (lambda ()
           (Success (computation)))])
      
      (lambda (error)
        (Error (format "Task execution failed: ~a" error))))))

;; Cluster coordinator
(define cluster-coordinator : (Behavior)
  (lambda ()
    (letrec ([cluster-nodes (make-hash-map)]
             [task-queue (make-queue)]
             [active-tasks (make-hash-map)]
             
             [handle-message
               (lambda (message sender state)
                 (match message
                   [(NodeJoin node-id address) =>
                    (lambda ()
                      (log-info (format "Coordinator: Node ~a joined" node-id))
                      (hash-map-set! cluster-nodes node-id address)
                      ;; Broadcast to all nodes
                      (broadcast-to-cluster (NodeJoin node-id address))
                      ;; Assign pending tasks
                      (assign-pending-tasks!)
                      state)]
                   
                   [(WorkRequest id task requester) =>
                    (lambda ()
                      ;; Add to task queue
                      (queue-enqueue! task-queue (WorkRequest id task requester))
                      (assign-pending-tasks!)
                      state)]
                   
                   [(WorkResponse id result) =>
                    (lambda ()
                      ;; Task completed, clean up
                      (hash-map-remove! active-tasks id)
                      (log-info (format "Coordinator: Task ~a completed" id))
                      state)]
                   
                   [_ =>
                    (lambda ()
                      state)]))]
             
             [broadcast-to-cluster
               (lambda (message)
                 (hash-map-for-each
                   cluster-nodes
                   (lambda (node-id address)
                     (let ([node-actor (remote-actor node-id address)])
                       (send! node-actor message)))))]
             
             [assign-pending-tasks!
               (lambda ()
                 (while (and (not (queue-empty? task-queue))
                           (not (hash-map-empty? cluster-nodes)))
                   (let ([work-request (queue-dequeue! task-queue)]
                         [target-node (select-best-node)])
                     (when target-node
                       (match work-request
                         [(WorkRequest id task requester) =>
                          (lambda ()
                            (send! target-node work-request)
                            (hash-map-set! active-tasks id target-node)
                            (log-info (format "Assigned task ~a to ~a"
                                             id (actor-id target-node))))])))))]
             
             [select-best-node
               (lambda ()
                 ;; Round-robin or load-based selection
                 (if (hash-map-empty? cluster-nodes)
                     #f
                     (let ([nodes (hash-map-keys cluster-nodes)])
                       (list-ref nodes (modulo (current-timestamp) (length nodes))))))])
      
      (behavior handle-message (hash-map)))))

;; High-level API for distributed computation
(define distributed-map : (∀ α β. (α → β) → (List α) → Distributed (List β))
  (lambda (mapper data)
    (do-distributed
      [coordinator ← (spawn-actor cluster-coordinator)]
      [task-id ← (generate-uuid)]
      [chunks (partition-data data (cluster-size))]
      [results ← (parallel-map
                   (lambda (chunk)
                     (do-distributed
                       [result-promise ← (make-promise)]
                       [_ ← (send! coordinator
                                  (WorkRequest (string-append task-id "-" (uuid))
                                              (MapTask chunk mapper)
                                              self))]
                       [result ← (await result-promise)]
                       (return result)))
                   chunks)]
      (return (concatenate results)))))

;; Usage example: Distributed prime number computation
(define find-primes-distributed : (Integer → Integer → Distributed (List Integer))
  (lambda ((start : Integer) (end : Integer))
    (let ([numbers (range start end)])
      (distributed-map
        (lambda (n)
          (if (is-prime? n) (list n) '()))
        numbers))))

;; Main function to start distributed system
(define start-distributed-node : (String → Integer → IO Unit)
  (lambda ((node-id : String) (port : Integer))
    (do-io
      [address ← (local-network-address port)]
      [node ← (spawn-actor (node-actor node-id address))]
      [_ ← (log-info (format "Started node ~a on port ~a" node-id port))]
      
      ;; Start heartbeat sender
      [_ ← (spawn-periodic
             (lambda ()
               (broadcast-heartbeat node-id))
             5000)]  ; Every 5 seconds
      
      ;; Join cluster
      [coordinator-address ← (resolve-coordinator)]
      [coordinator ← (remote-actor "coordinator" coordinator-address)]
      [_ ← (send! coordinator (NodeJoin node-id address))]
      
      ;; Keep node alive
      (actor-system-await-termination))))

;; High-frequency trading system with risk management
(import (scheme base)

;; Financial instrument types
(define-type Instrument

  (Option (underlying : String) (strike : Real) (expiry : Date) (type : OptionType))

;; Market data
(define-type MarketData
  (MarketData (instrument : Instrument)
             (timestamp : Timestamp)
             (bid : Real)
             (ask : Real)
             (last : Real)
             (volume : Integer)))

;; Order types
(define-type Order
  (MarketOrder (instrument : Instrument) (quantity : Integer) (side : Side))
  (LimitOrder (instrument : Instrument) (quantity : Integer) (side : Side) (price : Real))
  (StopOrder (instrument : Instrument) (quantity : Integer) (side : Side) (stop-price : Real)))

;; Portfolio tracking with dependent types
(define-type Portfolio

            (cash : Real)

(define-type Position

           (average-cost : Real)
           (market-value : Real)

;; Risk management system
(define-type RiskLimits
  (RiskLimits (max-position-size : Real)
             (max-daily-loss : Real)
             (max-leverage : Real)
             (restricted-instruments : (Set Instrument))))

;; Trading strategy interface
(define-type Strategy
  (Strategy (name : String)
           (parameters : (Map String Dynamic))
           (generate-signals : (TimeSeries MarketData → List Signal))
           (risk-limits : RiskLimits)))

;; Trading signals
(define-type Signal
  (BuySignal (instrument : Instrument) (confidence : Real) (target-price : Real))
  (SellSignal (instrument : Instrument) (confidence : Real) (target-price : Real))
  (HoldSignal (instrument : Instrument)))

;; Moving average crossover strategy
(define moving-average-strategy : (Integer → Integer → Strategy)
  (lambda ((short-period : Integer) (long-period : Integer))
    (Strategy
      "MovingAverageCrossover"
      (hash-map `(("short_period" . ,short-period)
                  ("long_period" . ,long-period)))
      
      ;; Signal generation function
      (lambda ((price-data : TimeSeries MarketData))
        (let ([prices (time-series-map (lambda (data) (last data)) price-data)]
              [short-ma (moving-average prices short-period)]
              [long-ma (moving-average prices long-period)])
          
          ;; Generate crossover signals
          (time-series-zip-with
            (lambda (short long prev-short prev-long price-point)
              (let ([instrument (instrument (time-series-value price-point))])
                (cond
                  ;; Golden cross: short MA crosses above long MA
                  [(and (> short long) (<= prev-short prev-long))
                   (BuySignal instrument 0.8 (last (time-series-value price-point)))]
                  
                  ;; Death cross: short MA crosses below long MA
                  [(and (< short long) (>= prev-short prev-long))
                   (SellSignal instrument 0.8 (last (time-series-value price-point)))]
                  
                  ;; No signal
                  [else (HoldSignal instrument)])))
            short-ma
            long-ma
            (time-series-lag short-ma 1)
            (time-series-lag long-ma 1)
            price-data)))
      
      ;; Risk limits for this strategy
      (RiskLimits 100000.0   ; Max $100k position
                 10000.0     ; Max $10k daily loss
                 2.0         ; Max 2x leverage
                 (set)))))   ; No restricted instruments

;; Risk manager
(define risk-manager : (Portfolio → RiskLimits → Order → Either String Order)
  (lambda ((portfolio : Portfolio) (limits : RiskLimits) (order : Order))
    (let ([instrument (get-order-instrument order)]
          [quantity (get-order-quantity order)]
          [side (get-order-side order)])
      
      ;; Check position size limits
      (let ([current-position (hash-map-ref (positions portfolio) instrument
                                           (Position 0 0.0 0.0 0.0))]
            [new-quantity (if (eq? side 'Buy)
                             (+ (quantity current-position) quantity)
                             (- (quantity current-position) quantity))])
        
        (cond
          ;; Position size check
          [(> (abs (* new-quantity (get-market-price instrument)))
              (max-position-size limits))
           (Left "Position size limit exceeded")]
          
          ;; Daily loss check
          [(> (portfolio-daily-loss portfolio) (max-daily-loss limits))
           (Left "Daily loss limit exceeded")]
          
          ;; Leverage check
          [(> (portfolio-leverage portfolio) (max-leverage limits))
           (Left "Leverage limit exceeded")]
          
          ;; Restricted instrument check
          [(set-contains? (restricted-instruments limits) instrument)
           (Left "Instrument is restricted")]
          
          ;; All checks passed
          [else (Right order)])))))

;; Order execution engine
(define order-executor : (Order → MarketData → IO ExecutionResult)
  (lambda ((order : Order) (market-data : MarketData))
    (match order
      [(MarketOrder instrument quantity side) =>
       (lambda ()
         (let ([execution-price (if (eq? side 'Buy)
                                   (ask market-data)
                                   (bid market-data))])
           (do-io
             [execution-id ← (generate-uuid)]
             [timestamp ← (current-timestamp)]
             [_ ← (log-trade execution-id instrument quantity side execution-price)]
             (return (ExecutionResult execution-id
                                    instrument
                                    quantity
                                    side
                                    execution-price
                                    timestamp)))))]
      
      [(LimitOrder instrument quantity side limit-price) =>
       (lambda ()
         (let ([market-price (if (eq? side 'Buy) (ask market-data) (bid market-data))])
           (if (or (and (eq? side 'Buy) (<= market-price limit-price))
                   (and (eq? side 'Sell) (>= market-price limit-price)))
               ;; Execute at limit price
               (do-io
                 [execution-id ← (generate-uuid)]
                 [timestamp ← (current-timestamp)]
                 [_ ← (log-trade execution-id instrument quantity side limit-price)]
                 (return (ExecutionResult execution-id
                                        instrument
                                        quantity
                                        side
                                        limit-price
                                        timestamp)))
               ;; Order not executable
               (return (PendingOrder order)))))]
      
      [_ =>
       (lambda ()
         (return (OrderRejected "Unsupported order type")))])))

;; Portfolio manager
(define portfolio-manager : (Behavior)
  (lambda ()
    (letrec ([portfolio (Portfolio (hash-map) 1000000.0 1000000.0 (current-timestamp))]
             [pending-orders (make-queue)]
             
             [handle-message
               (lambda (message sender state)
                 (match message
                   [(MarketData-Update data) =>
                    (lambda ()
                      ;; Update portfolio valuation
                      (let ([updated-portfolio (update-portfolio-valuation portfolio data)])
                        ;; Check pending orders
                        (process-pending-orders! data)
                        updated-portfolio))]
                   
                   [(OrderRequest order) =>
                    (lambda ()
                      ;; Risk check
                      (match (risk-manager portfolio (get-strategy-limits) order)
                        [(Right approved-order) =>
                         (lambda ()
                           (queue-enqueue! pending-orders approved-order)
                           (send! sender (OrderAccepted (generate-uuid)))
                           portfolio)]
                        [(Left reason) =>
                         (lambda ()
                           (send! sender (OrderRejected reason))
                           portfolio)]))]
                   
                   [(ExecutionReport execution) =>
                    (lambda ()
                      ;; Update portfolio with execution
                      (update-portfolio-with-execution portfolio execution))]
                   
                   [_ =>
                    (lambda () portfolio)]))]
             
             [process-pending-orders!
               (lambda (market-data)
                 (let ([orders-to-process (queue-drain! pending-orders)])
                   (for-each
                     (lambda (order)
                       (spawn-task
                         (lambda ()
                           (do-io
                             [result ← (order-executor order market-data)]
                             (match result
                               [(ExecutionResult _ _ _ _ _ _) =>
                                (lambda ()
                                  (send! self (ExecutionReport result)))]
                               [(PendingOrder pending) =>
                                (lambda ()
                                  (queue-enqueue! pending-orders pending))]
                               [_ =>
                                (lambda ()
                                  (log-warning (format "Order failed: ~a" result)))])))))
                     orders-to-process)))])
      
      (behavior handle-message portfolio))))

;; Market data feed processor
(define market-data-processor : (String → Behavior)
  (lambda ((feed-url : String))
    (letrec ([subscribers (make-list)]
             [price-cache (make-hash-map)]
             
             [handle-message
               (lambda (message sender state)
                 (match message
                   [(Subscribe) =>
                    (lambda ()
                      (set! subscribers (cons sender subscribers))
                      state)]
                   
                   [(MarketData-Raw raw-data) =>
                    (lambda ()
                      ;; Parse and validate market data
                      (match (parse-market-data raw-data)
                        [(Some market-data) =>
                         (lambda ()
                           ;; Update cache
                           (hash-map-set! price-cache
                                         (instrument market-data)
                                         market-data)
                           ;; Broadcast to subscribers
                           (for-each
                             (lambda (subscriber)
                               (send! subscriber (MarketData-Update market-data)))
                             subscribers)
                           state)]
                        [None =>
                         (lambda ()
                           (log-warning "Invalid market data received")
                           state)]))]
                   
                   [_ =>
                    (lambda () state)]))])
      
      ;; Start market data feed
      (spawn-task
        (lambda ()
          (connect-market-feed feed-url
                              (lambda (raw-data)
                                (send! self (MarketData-Raw raw-data))))))
      
      (behavior handle-message (hash-map)))))

;; Main trading system
(define start-trading-system : (IO Unit)
  (lambda ()
    (do-io
      [_ ← (log-info "Starting trading system...")]
      
      ;; Start components
      [market-processor ← (spawn-actor (market-data-processor "ws://market-feed.example.com"))]
      [portfolio-mgr ← (spawn-actor (portfolio-manager))]
      [strategy ← (moving-average-strategy 20 50)]
      
      ;; Subscribe portfolio manager to market data
      [_ ← (send! market-processor (Subscribe))]
      
      ;; Start strategy execution loop
      [_ ← (spawn-periodic
             (lambda ()
               (do-io
                 [signals ← (generate-strategy-signals strategy)]
                 (for-each
                   (lambda (signal)
                     (match signal
                       [(BuySignal instrument confidence target-price) =>
                        (lambda ()
                          (let ([order (MarketOrder instrument 100 'Buy)])
                            (send! portfolio-mgr (OrderRequest order))))]
                       [(SellSignal instrument confidence target-price) =>
                        (lambda ()
                          (let ([order (MarketOrder instrument 100 'Sell)])
                            (send! portfolio-mgr (OrderRequest order))))]
                       [_ => (lambda () unit)]))
                   signals)))
             1000)]  ; Every second
      
      [_ ← (log-info "Trading system started successfully")]
      (actor-system-await-termination))))
