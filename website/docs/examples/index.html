Fibonacci with Type Evolution

See how a simple function evolves from dynamic to fully typed:

Dynamic to Static Evolution

;; Stage 1: Dynamic implementation
(define fibonacci
  (lambda (n)
    (if (< n 2) n
        (+ (fibonacci (- n 1))
           (fibonacci (- n 2))))))

;; Stage 2: Add input type
(define fibonacci
  (lambda ((n : Integer))
    (if (< n 2) n
        (+ (fibonacci (- n 1))
           (fibonacci (- n 2))))))

;; Stage 3: Full type annotation
(define fibonacci : (Integer → Integer)
  (lambda ((n : Integer))
    (if (< n 2) n
        (+ (fibonacci (- n 1))
           (fibonacci (- n 2))))))

;; Stage 4: Optimized with memoization
(define fibonacci-memo : (Integer → State (Map Integer Integer) Integer)
  (lambda ((n : Integer))
    (do-state
      [cache ← get-state]
      (match (map-lookup cache n)
        [(Some result) (return result)]
        [None 
         (if (< n 2) 
             (do-state
               [_ ← (modify-state (lambda (m) (map-insert m n n)))]
               (return n))
             (do-state
               [f1 ← (fibonacci-memo (- n 1))]
               [f2 ← (fibonacci-memo (- n 2))]
               [result ← (pure (+ f1 f2))]
               [_ ← (modify-state (lambda (m) (map-insert m n result)))]
               (return result)))]))))

Safe Matrix Operations with Dependent Types

Compile-time dimension checking for matrix operations:

Type-Safe Linear Algebra

;; Matrix type with dimensions in the type
(define-type (Matrix rows cols T)
  (matrix-with-dimensions rows cols T))

;; Safe matrix creation
(define make-matrix : (∀ (r c : Nat) (T : Type) 
                        → T → Matrix r c T)
  (lambda (default-value)
    (create-matrix r c default-value)))

;; Matrix multiplication with dimension constraints
(define matrix-multiply : (∀ (a b c : Nat) (T : Type)
                            → Matrix a b T 
                            → Matrix b c T 
                            → Matrix a c T)
  (lambda (m1 m2)
    (let ([result (make-matrix (zero T))])
      (for-range 0 a (lambda (i)
        (for-range 0 c (lambda (j)
          (let ([sum (fold-range 0 b (lambda (k acc)
                       (+ acc (* (matrix-ref m1 i k)
                                (matrix-ref m2 k j))))
                     (zero T))])
            (matrix-set! result i j sum)))))
      result)))

;; Usage - this compiles successfully
(let ([m1 : (Matrix 3 4 Real) (make-matrix 1.0)]
      [m2 : (Matrix 4 2 Real) (make-matrix 2.0)])
  (matrix-multiply m1 m2))  ; Result: Matrix 3 2 Real

;; This would be a compile-time error:
;; (matrix-multiply m1 (make-matrix 3 3 2.0))  ; Error: 4 ≠ 3

File Processing with Effect Handling

Safe file operations with comprehensive error handling:

Robust File Processing

;; Define custom error types
(define-type FileError
  (FileNotFound String)
  (PermissionDenied String)
  (InvalidFormat String))

;; Safe file reading with error handling
(define read-config-file : (String → IO (Either FileError Config))
  (lambda (filename)
    (with-handler
      [(IOError (FileNotFound path)) → 
       (return (Left (FileNotFound path)))]
      [(IOError (PermissionDenied path)) → 
       (return (Left (PermissionDenied path)))]
      [(ParseError msg) → 
       (return (Left (InvalidFormat msg)))]
      
      (do-io
        [exists? ← (file-exists? filename)]
        (if (not exists?)
            (return (Left (FileNotFound filename)))
            (do-io
              [content ← (read-file filename)]
              [config ← (parse-config content)]
              (return (Right config))))))))

;; Process multiple configuration files
(define process-configs : (List String → IO (Either FileError (List Config)))
  (lambda (filenames)
    (do-io
      [results ← (map-io read-config-file filenames)]
      (return (sequence-either results)))))

;; Usage with proper error reporting
(define main : (IO Unit)
  (lambda ()
    (do-io
      [result ← (process-configs '("app.conf" "db.conf" "cache.conf"))]
      (match result
        [(Left (FileNotFound path)) 
         (println (format "Error: Configuration file not found: ~a" path))]
        [(Left (PermissionDenied path))
         (println (format "Error: Permission denied for file: ~a" path))]
        [(Left (InvalidFormat msg))
         (println (format "Error: Invalid configuration format: ~a" msg))]
        [(Right configs)
         (do-io
           [_ ← (println "All configuration files loaded successfully")]
           (start-application configs))]))))

Learning Path

We recommend exploring examples in this order:

  1. Basic Programs - Start with simple syntax and concepts
  2. Type System - Learn gradual typing and type evolution
  3. Effect System - Understand monadic programming
  4. Data Structures - Implement functional data structures
  5. Real-world Applications - Complete programs and applications

Running Examples

All examples can be run directly in the Lambdust REPL or saved as .ldust files:

# Run in REPL
$ lambdust
λ> (define fibonacci (lambda (n) ...))
λ> (fibonacci 10)

# Run from file
$ echo '(define main (lambda () (println "Hello, World!")))' > hello.ldust
$ lambdust hello.ldust

# Load in REPL
λ> :load examples/fibonacci.ldust
λ> (fibonacci 20)
