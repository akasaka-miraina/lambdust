Monadic Purity

Lambdust maintains referential transparency through monadic purity, where side effects are explicitly tracked in the type system and managed through monadic composition. This approach provides the mathematical elegance of pure functional programming while enabling practical interaction with the real world.

M(A) >>= (A → M(B)) → M(B) — Monadic bind operation

Pure vs. Effectful Functions

In Lambdust, functions are pure by default. Side effects must be explicitly declared in the type signature:

Pure vs. Effectful

;; Pure function - no side effects
(define add : (Integer → Integer → Integer)
  (lambda ((x : Integer) (y : Integer))
    (+ x y)))

;; Effectful function - I/O side effects
(define greet : (String → IO Unit)
  (lambda ((name : String))
    (println (string-append "Hello, " name "!"))))

;; Mixed - pure computation, effectful result
(define compute-and-log : (Integer → IO Integer)
  (lambda ((n : Integer))
    (let ([result (* n n)])  ; Pure computation
      (do-io
        [_ ← (println (format "Computed: ~a" result))]  ; Effect
        (return result)))))

The IO Monad

Input/output operations are wrapped in the IO monad to maintain purity:

;; Basic I/O operations
(define read-line : (IO String))
(define print : (String → IO Unit))
(define read-file : (String → IO String))
(define write-file : (String → String → IO Unit))

;; Composing I/O operations
(define interactive-greeting : (IO Unit)
  (lambda ()
    (do-io
      [_ ← (print "What's your name? ")]
      [name ← (read-line)]
      [_ ← (print (string-append "Hello, " name "!"))]
      (return unit))))

Do-notation for Monadic Composition

The do-io syntax provides convenient monadic composition:

;; File processing with error handling
(define process-file : (String → String → IO (Either String Unit))
  (lambda (input-file output-file)
    (do-io
      [content ← (read-file input-file)]
      [processed ← (pure (string-upcase content))]
      [_ ← (write-file output-file processed)]
      [_ ← (println "File processed successfully")]
      (return (Right unit)))))

;; Desugars to:
(define process-file-desugared : (String → String → IO (Either String Unit))
  (lambda (input-file output-file)
    (>>= (read-file input-file)
         (lambda (content)
           (>>= (pure (string-upcase content))
                (lambda (processed)
                  (>>= (write-file output-file processed)
                       (lambda (_)
                         (>>= (println "File processed successfully")
                              (lambda (_)
                                (return (Right unit))))))))))

State Monad

Mutable state is handled through the State monad:

Stateful Computation

;; Define state type
(define-type Counter Integer)

;; State operations
(define get-counter : (State Counter Counter)
  (lambda () get-state))

(define increment-counter : (State Counter Unit)
  (lambda ()
    (do-state
      [current ← get-state]
      [_ ← (put-state (+ current 1))]
      (return unit))))

;; Stateful computation
(define count-items : (List α → State Counter Integer)
  (lambda (items)
    (do-state
      [_ ← (put-state 0)]  ; Initialize counter
      [_ ← (map-state (lambda (item)
                        (do-state
                          [_ ← increment-counter]
                          (return unit)))
                      items)]
      [final ← get-state]
      (return final))))

;; Running stateful computation
(define example : Integer
  (run-state (count-items '(a b c d e)) 0))  ; Result: 5

Error Handling with Monads

Error handling is integrated into the monadic system:

;; Either monad for error handling
(define safe-divide : (Real → Real → Either String Real)
  (lambda (x y)
    (if (= y 0)
        (Left "Division by zero")
        (Right (/ x y)))))

;; Maybe monad for optional values
(define safe-head : (∀ α. List α → Maybe α)
  (lambda (lst)
    (if (null? lst)
        (None)
        (Some (car lst)))))

;; Combining with I/O
(define safe-file-operation : (String → IO (Either String String))
  (lambda (filename)
    (with-handler
      [(IOError msg) → (return (Left msg))]
      (do-io
        [content ← (read-file filename)]
        [result ← (pure (process-content content))]
        (return (Right result))))))

Effect Polymorphism

Functions can be polymorphic over effects:

;; Generic logging function
(define log-message : (∀ (m : Monad). String → m Unit)
  (lambda (msg)
    (case (monad-type m)
      [IO → (println msg)]
      [State → (modify-state (lambda (s) (cons msg s)))]
      [Writer → (tell msg)])))

;; Effect-polymorphic map
(define map-effect : (∀ α β (m : Monad). (α → m β) → List α → m (List β))
  (lambda (f lst)
    (match lst
      [() → (return '())]
      [(cons x xs) → 
       (do-monad
         [y ← (f x)]
         [ys ← (map-effect f xs)]
         (return (cons y ys)))])))

;; Usage with different effects
(map-effect (lambda (x) (println x)) '(1 2 3))     ; IO effect
(map-effect (lambda (x) (increment-counter)) '(1 2 3))  ; State effect

Algebraic Effects

Lambdust supports algebraic effects for more flexible effect handling:

Custom Effects

;; Define custom effects
(define-effect Logging
  (log : String → Unit))

(define-effect Database  
  (query : String → List Row)
  (insert : Row → Unit))

;; Function using multiple effects
(define process-user : (UserId → Eff [Logging Database] User)
  (lambda (user-id)
    (do-eff
      [_ ← (log (format "Processing user: ~a" user-id))]
      [rows ← (query (format "SELECT * FROM users WHERE id = ~a" user-id))]
      (match rows
        [() → (throw (UserNotFound user-id))]
        [(cons row _) → 
         (do-eff
           [user ← (pure (parse-user row))]
           [_ ← (log (format "User loaded: ~a" (user-name user)))]
           (return user))]))))

;; Effect handlers
(define with-console-logging : (∀ α. Eff [Logging] α → IO α)
  (lambda (computation)
    (handle computation
      [(log msg) → (do-io
                     [_ ← (println msg)]
                     (resume unit))])))

(define with-mock-database : (∀ α. Eff [Database] α → IO α)
  (lambda (computation)
    (handle computation
      [(query sql) → (resume mock-data)]
      [(insert row) → (do-io
                        [_ ← (println "Mock insert")]
                        (resume unit))])))

Monad Transformers

Combine multiple monadic effects using monad transformers:

;; ReaderT for configuration
(define-type Config
  (Config String Integer Boolean))

;; StateT for mutable state  
(define-type AppState
  (AppState Integer (Map String String)))

;; Combined effect stack
(define-type App α
  (ReaderT Config (StateT AppState IO) α))

;; Function using the combined stack
(define run-app-logic : (String → App String)
  (lambda (input)
    (do-app
      [config ← ask]  ; From ReaderT
      [state ← get]   ; From StateT
      [_ ← (liftIO (println "Processing..."))]  ; From IO
      [_ ← (modify (lambda (s) (update-app-state s input)))]
      (return (process-with-config config input)))))

;; Running the computation
(define main : (IO Unit)
  (lambda ()
    (let ([config (Config "prod" 8080 #t)]
          [initial-state (AppState 0 (empty-map))])
      (do-io
        [result ← (run-state-t (run-reader-t (run-app-logic "test") config) 
                              initial-state)]
        (println (format "Result: ~a" result))))))

Performance Considerations

Zero-Cost Abstractions

  • Compile-time optimization: Monadic code is optimized away when possible
  • Effect inference: The compiler can infer and eliminate unnecessary effect wrappers
  • Specialization: Monomorphic instances are generated for performance
  • Inlining: Small monadic functions are inlined automatically

Performance Tips

  • Prefer specific monads (IO, State) over general effect systems when possible
  • Use pure to lift pure computations rather than wrapping in effects
  • Batch I/O operations to minimize effect overhead
  • Consider using unsafe-perform-io for proven-pure computations

Integration with Type System

Monadic purity works seamlessly with Lambdust's advanced type features:

;; Dependent types with effects
(define read-vector : (∀ (n : Nat) (T : Type). 
                       String → IO (Vector n T))
  (lambda (filename)
    (do-io
      [content ← (read-file filename)]
      [values ← (parse-values content)]
      (if (= (length values) n)
          (return (list→vector values))
          (throw (InvalidVectorSize n (length values)))))))

;; Gradual typing with effects  
(define process-data
  (lambda (input)  ; Inferred: Dynamic → IO Dynamic
    (do-io
      [result ← (compute-something input)]
      [_ ← (log-result result)]
      (return result))))

;; Gradually add types
(define process-data-typed : (Config → IO Result)
  (lambda ((config : Config))
    (do-io
      [result ← (compute-something-typed config)]
      [_ ← (log-result result)]
      (return result))))

Best Practices

  1. Keep pure code pure: Separate pure computations from effectful operations
  2. Use specific monads: Prefer IO, State, etc. over generic effect systems
  3. Minimize effect scope: Keep effectful regions as small as possible
  4. Handle errors explicitly: Use Either or exception handling for error cases
  5. Compose effects: Use do-notation for readable monadic composition
  6. Document effects: Make side effects clear in function types and documentation
